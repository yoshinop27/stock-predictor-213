void linearRegression(double prices[], int numSize, double *slope, double *yint)
{
double sumX = 0;
double sumY = 0;
double sumXY = 0;
double sumXSq = 0;

// Go through all values to find the sums
// x is days y = price
for (int i = 0; i < numSize; i++)
{
sumX = sumX + i;
sumY = sumY + prices[i];
sumXY = sumXY + (prices[i] * i);
sumXSq = sumXSq + (i * i);
}

// calculate slope
*slope = (numSize * sumXY - sumX * sumY) / (numSize * sumXSq - sumX * sumX);

// calculate y-int
*yint = (sumY - (*slope) * sumX) / numSize;
}
